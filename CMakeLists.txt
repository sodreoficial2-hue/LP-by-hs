 if(NOT LP_WEBVIEW2_SDK_ROOT)
  set(LP_WEBVIEW2_SDK_ROOT "${CMAKE_BINARY_DIR}/_deps/webview2")
  if(NOT EXISTS "${LP_WEBVIEW2_SDK_ROOT}/build/native/include/WebView2.h")
   file(MAKE_DIRECTORY "${LP_WEBVIEW2_SDK_ROOT}")
   file(DOWNLOAD "https://api.nuget.org/v3-flatcontainer/microsoft.web.webview2/1.0.2903.40/microsoft.web.webview2.1.0.2903.40.nupkg"
    "${LP_WEBVIEW2_SDK_ROOT}/sdk.zip" EXPECTED_HASH SHA256=ef128016dd1e51c59178c827ed5b8aa3322c57afa8675d930f8109505542ad74 TLS_VERIFY ON)
   execute_process(COMMAND "${CMAKE_COMMAND}" -E tar xf sdk.zip WORKING_DIRECTORY "${LP_WEBVIEW2_SDK_ROOT}" RESULT_VARIABLE lp_webview_extract)
   if(NOT lp_webview_extract EQUAL 0)
    message(FATAL_ERROR "Cannot extract the pinned WebView2 SDK")
   endif()
  endif()
 endif()
 if(NOT EXISTS "${LP_WEBVIEW2_SDK_ROOT}/build/native/include/WebView2.h")
  message(FATAL_ERROR "LP_WEBVIEW2_SDK_ROOT must point to the extracted NuGet package")
 endif()
endif()

  endif()
  target_link_libraries(LivePerformanceVision PRIVATE lp_audio winmm comctl32 comdlg32 gdi32 user32 shell32 ole32 uuid gdiplus)
  if(MSVC)
   target_link_options(LivePerformanceVision PRIVATE /MANIFEST:NO)
  elseif(MINGW)
   target_link_options(LivePerformanceVision PRIVATE -municode -static)
  endif()
  add_custom_command(TARGET LivePerformanceVision POST_BUILD
   COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/vision-ui" "$<TARGET_FILE_DIR:LivePerformanceVision>/vision-ui"
   COMMAND ${CMAKE_COMMAND} -E copy_if_different "${LP_WEBVIEW2_SDK_ROOT}/build/native/x64/WebView2Loader.dll" "$<TARGET_FILE_DIR:LivePerformanceVision>")
  if(LP_COPY_RECOVERED_ASSETS)

 endif()
