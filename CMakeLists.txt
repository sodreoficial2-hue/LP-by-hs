cmake_minimum_required(VERSION 3.16)
project(LPbyHS LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(lp_engine src/sfizz_engine.cpp src/sf2_engine.cpp src/dsp.cpp)
target_include_directories(lp_engine PUBLIC src)
target_link_libraries(lp_engine PRIVATE ${CMAKE_DL_LIBS})
add_executable(lp_catalog src/main.cpp)
add_executable(lp_session_test tests/session_test.cpp)
target_include_directories(lp_session_test PRIVATE src)
add_library(lp_fake_sfizz SHARED tests/fake_sfizz.cpp)
add_executable(lp_engine_midi_test tests/engine_midi_test.cpp)
target_link_libraries(lp_engine_midi_test PRIVATE lp_engine Threads::Threads)
option(LP_TEST_ONLY "Build regression tests without Windows ASIO SDK" OFF)
if(WIN32 AND NOT LP_TEST_ONLY)
 set(LP_PORTAUDIO_SOURCE "" CACHE PATH "PortAudio v19.7.0 source directory")
 set(ASIOSDK_ROOT_DIR "" CACHE PATH "Steinberg ASIO SDK directory containing common/asio.h")
 if(NOT EXISTS "${LP_PORTAUDIO_SOURCE}/CMakeLists.txt" OR NOT EXISTS "${ASIOSDK_ROOT_DIR}/common/asio.h")
  message(FATAL_ERROR "Windows build requires LP_PORTAUDIO_SOURCE and ASIOSDK_ROOT_DIR. See docs/RECONSTRUCAO-ETAPA-4.md")
 endif()
 set(PA_USE_ASIO ON CACHE BOOL "" FORCE)
 foreach(backend DS WMME WASAPI WDMKS)
  set(PA_USE_${backend} OFF CACHE BOOL "" FORCE)
 endforeach()
 set(PA_BUILD_SHARED OFF CACHE BOOL "" FORCE)
 set(PA_BUILD_STATIC ON CACHE BOOL "" FORCE)
 set(PA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 set(PA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 add_subdirectory("${LP_PORTAUDIO_SOURCE}" "${CMAKE_BINARY_DIR}/portaudio" EXCLUDE_FROM_ALL)
 add_library(lp_audio src/audio_host.cpp)
 target_include_directories(lp_audio PUBLIC src third_party/portaudio)
 target_link_libraries(lp_audio PUBLIC lp_engine portaudio_static)
 enable_language(RC)
 add_executable(LivePerformance WIN32 src/win32_app.cpp src/windows.rc)
 # windows.rc already embeds app.manifest as RT_MANIFEST (resource 1).
 # Disable MSVC's automatic second manifest to prevent CVT1100/LNK1123.
 if(MSVC)
  target_link_options(LivePerformance PRIVATE /MANIFEST:NO)
 endif()
 target_include_directories(LivePerformance PRIVATE src)
 target_link_libraries(LivePerformance PRIVATE lp_audio winmm comctl32 comdlg32 gdi32 user32 shell32 ole32 uuid)
 if(MINGW)
  target_link_options(LivePerformance PRIVATE -municode -static)
 endif()
 set(LP_SOUNDBANK_ROOT "" CACHE PATH "Assembled curated bank containing catalog.txt")
 if(LP_SOUNDBANK_ROOT)
  if(NOT EXISTS "${LP_SOUNDBANK_ROOT}/catalog.txt")
   message(FATAL_ERROR "LP_SOUNDBANK_ROOT must contain catalog.txt")
  endif()
  add_custom_command(TARGET LivePerformance POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${LP_SOUNDBANK_ROOT}" "$<TARGET_FILE_DIR:LivePerformance>/soundbank")
 endif()
 option(LP_COPY_RECOVERED_ASSETS "Copy recovered sfizz runtime next to Windows app" ON)
 if(LP_COPY_RECOVERED_ASSETS)
  foreach(runtime sfizz.dll sndfile.dll SDL3.dll)
   add_custom_command(TARGET LivePerformance POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/recovery/extracted/live/${runtime}" "$<TARGET_FILE_DIR:LivePerformance>")
  endforeach()
  add_custom_command(TARGET LivePerformance POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/recovery/extracted/live/samples" "$<TARGET_FILE_DIR:LivePerformance>/samples")
 endif()
endif()
enable_testing()
add_test(NAME session_regression COMMAND lp_session_test "${CMAKE_CURRENT_SOURCE_DIR}/recovery/extracted/live/samples")
add_test(NAME engine_midi_regression COMMAND lp_engine_midi_test $<TARGET_FILE:lp_fake_sfizz>)

add_executable(lp_dsp_test tests/dsp_test.cpp src/dsp.cpp)
target_include_directories(lp_dsp_test PRIVATE src)
add_test(NAME dsp_regression COMMAND lp_dsp_test)

add_executable(lp_audio_host_test tests/audio_host_test.cpp tests/fake_portaudio.cpp src/audio_host.cpp)
target_include_directories(lp_audio_host_test PRIVATE src third_party/portaudio)
target_link_libraries(lp_audio_host_test PRIVATE lp_engine Threads::Threads)
add_test(NAME audio_host_regression COMMAND lp_audio_host_test $<TARGET_FILE:lp_fake_sfizz>)

set(LP_BANK_TEST_ROOT "" CACHE PATH "Optional installed soundbank for real render validation")
set(LP_REAL_SFIZZ_LIBRARY "" CACHE FILEPATH "Optional real sfizz library for offline render integration tests")
if(LP_REAL_SFIZZ_LIBRARY)
 add_executable(lp_real_performance_test tests/real_performance_test.cpp)
 target_link_libraries(lp_real_performance_test PRIVATE lp_engine)
 add_test(NAME real_performance_controls COMMAND lp_real_performance_test "${LP_REAL_SFIZZ_LIBRARY}" "${LP_BANK_TEST_ROOT}")
 add_executable(lp_real_sfizz_envelope_test tests/real_sfizz_envelope_test.cpp)
 target_link_libraries(lp_real_sfizz_envelope_test PRIVATE lp_engine)
 add_test(NAME real_sfizz_envelope COMMAND lp_real_sfizz_envelope_test "${LP_REAL_SFIZZ_LIBRARY}" "${CMAKE_CURRENT_SOURCE_DIR}/recovery/extracted/live/samples")
endif()

add_executable(lp_instrument_catalog_test tests/instrument_catalog_test.cpp)
target_include_directories(lp_instrument_catalog_test PRIVATE src)
add_test(NAME instrument_catalog_regression COMMAND lp_instrument_catalog_test)
if(LP_REAL_SFIZZ_LIBRARY AND LP_BANK_TEST_ROOT)
 add_executable(lp_bank_render_test tests/bank_render_test.cpp)
 target_link_libraries(lp_bank_render_test PRIVATE lp_engine)
 add_test(NAME real_bank_render COMMAND lp_bank_render_test "${LP_REAL_SFIZZ_LIBRARY}" "${LP_BANK_TEST_ROOT}")
 find_package(Python3 COMPONENTS Interpreter)
 if(Python3_Interpreter_FOUND)
  add_test(NAME real_bank_mono COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/bank_mono_test.py" "${LP_REAL_SFIZZ_LIBRARY}" "${LP_BANK_TEST_ROOT}")
 endif()
endif()

set(LP_REAL_FLUIDSYNTH "" CACHE FILEPATH "FluidSynth shared library for SF2 integration test")
set(LP_REAL_SF2 "" CACHE FILEPATH "Real SoundFont SF2 for integration test")
if(LP_REAL_FLUIDSYNTH AND LP_REAL_SF2)
 add_executable(lp_real_sf2_test tests/real_sf2_test.cpp)
 target_link_libraries(lp_real_sf2_test PRIVATE lp_engine)
 add_test(NAME real_sf2_render_adsr COMMAND lp_real_sf2_test "${LP_REAL_FLUIDSYNTH}" "${LP_REAL_SF2}")
endif()
